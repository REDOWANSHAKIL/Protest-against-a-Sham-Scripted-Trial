const upload = document.getElementById('upload'); const canvas = document.getElementById('canvas'); const ctx = canvas.getContext('2d'); const container = document.getElementById('canvas-container'); const downloadBtn = document.getElementById('download-btn'); const FRAME_SRC = 'frame.png'; let userImage = new Image(); let frameImage = new Image(); frameImage.src = FRAME_SRC; upload.addEventListener('change', (e) => { const file = e.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = (event) => { userImage.src = event.target.result; }; reader.readAsDataURL(file); }); userImage.onload = () => { ctx.clearRect(0, 0, canvas.width, canvas.height); // 1. Draw the main frame template first ctx.drawImage(frameImage, 0, 0, canvas.width, canvas.height); // 2. Create a circular mask exactly where the cutout is ctx.save(); ctx.beginPath(); // Center coordinates and radius matching your specific frame image ctx.arc(605, 440, 340, 0, Math.PI * 2); ctx.clip(); // 3. Draw the user image inside that circle cleanly let minDim = Math.min(userImage.width, userImage.height); let sx = (userImage.width - minDim) / 2; let sy = (userImage.height - minDim) / 2; ctx.drawImage(userImage, sx, sy, minDim, minDim, 265, 100, 680, 680); ctx.restore(); container.style.display = 'block'; }; downloadBtn.addEventListener('click', () => { const link = document.createElement('a'); link.download = 'framed-profile-pic.png'; link.href = canvas.toDataURL('image/png'); link.click(); });
